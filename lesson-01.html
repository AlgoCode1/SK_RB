1 Создай платформу и чекпоинт

Добавь в игру стартовую площадку и точку сохранения

📄 Скопируй в Script (ServerScriptService) 
-- Создаём стартовую платформу и чекпоинт
local platform = Instance.new("Part")
platform.Size = Vector3.new(20, 1, 20)
platform.Position = Vector3.new(0, 5, 0)
platform.Anchored = true
platform.BrickColor = BrickColor.new("Bright green")
platform.Parent = workspace

local checkpoint = Instance.new("Part")
checkpoint.Size = Vector3.new(4, 1, 4)
checkpoint.Position = Vector3.new(0, 10, -8)
checkpoint.Anchored = true
checkpoint.BrickColor = BrickColor.new("Gold")
checkpoint.Parent = workspace

print("Платформа и чекпоинт готовы!")

2 Скрипт для прыжка с ускорением

Увеличиваем силу прыжка игрока при нажатии на пробел

📄 Вставь в StarterCharacterScripts (LocalScript) 
-- Усиленный прыжок при нажатии пробела
local player = game.Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid")

local UIS = game:GetService("UserInputService")

UIS.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Space then
        humanoid.JumpPower = 100
        humanoid.Jump = true
        wait(1)
        humanoid.JumpPower = 50
    end
end)

print("Скрипт прыжка активирован!")

3 Сбор монет — счётчик очков

При касании монеты она исчезает, а игрок получает +1 очко

📄 Скопируй в Script внутри монеты (Part) 
-- Скрипт для монеты: сбор и начисление очков
local coin = script.Parent
local debounce = false

coin.Touched:Connect(function(hit)
    if debounce then return end
    local player = game.Players:GetPlayerFromCharacter(hit.Parent)
    if not player then return end

    debounce = true
    coin:Destroy()

    local leaderstats = player:FindFirstChild("leaderstats")
    if not leaderstats then
        leaderstats = Instance.new("Folder")
        leaderstats.Name = "leaderstats"
        leaderstats.Parent = player
    end

    local points = leaderstats:FindFirstChild("Очки")
    if not points then
        points = Instance.new("IntValue")
        points.Name = "Очки"
        points.Value = 0
        points.Parent = leaderstats
    end

    points.Value = points.Value + 1
    print(player.Name .. " собрал монету! Всего: " .. points.Value)
end)
